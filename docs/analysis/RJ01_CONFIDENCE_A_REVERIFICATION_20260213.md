```sql
-- ❌ 誤り: 枠番固定（1-2-3）
JOIN trifecta_odds t ON rp.race_id = t.race_id
    AND t.combination = '1-2-3'

-- ✅ 正しい: 予測ベース（現在使用中）
WITH prediction_combos AS (
    SELECT
        rp1.race_id,
        CAST(rp1.pit_number AS TEXT) || '-' ||
        CAST(rp2.pit_number AS TEXT) || '-' ||
        CAST(rp3.pit_number AS TEXT) as pred_combo
    FROM race_predictions rp1
    JOIN race_predictions rp2 ON rp1.race_id = rp2.race_id
        AND rp2.prediction_type = 'before' AND rp2.rank_prediction = 2
    JOIN race_predictions rp3 ON rp1.race_id = rp3.race_id
        AND rp3.prediction_type = 'before' AND rp3.rank_prediction = 3
    WHERE rp1.prediction_type = 'before' AND rp1.rank_prediction = 1
)
SELECT ... FROM prediction_combos pc
JOIN trifecta_odds t ON pc.race_id = t.race_id
    AND t.combination = pc.pred_combo
```

## ⚠️ リスクと注意点

### 1. サンプルサイズの制約

**50-100倍帯の懸念**:
- 全級別: 394件（サンプル不足）
- A1級限定: 2,518件（充分）

**リスク**: 50-100倍帯はサンプル不足のため、統計的信頼性が低い

**対策**: A1級フィルターで充分なサンプル確保

### 2. 的中率の低さ

**A1級×50-100倍の的中率**: 1.59%（40件/2,518件）

**リスク**: 極めて低い的中率のため、偶然的な黒字の可能性

**対策**: 黒字年数の厳格な確認（4/6年以上）

### 3. 市場効率化リスク

**問題**: 2022年以降のパフォーマンス悪化

**原因推測**:
- オッズマーケットの効率化
- A1級の相対的な優位性低下
- システムトレーダーの増加

**対策**: 継続的なモニタリング、定期的な再検証

### 4. 過学習リスク

**問題**: 過去データに最適化しすぎると、未来のデータで性能低下

**対策**:
- 年度別パフォーマンスの確認
- Out-of-sample テスト（新規データでの検証）
- 最小サンプル要件の厳守

## 付録: 分析に使用したSQLクエリ（参考）

### A信頼度×オッズ帯別パフォーマンス

```sql
-- 予測ベースの正確なオッズ取得
WITH prediction_combos AS (
    SELECT
        rp1.race_id,
        rp1.confidence_level,
        CAST(rp1.pit_number AS TEXT) || '-' ||
        CAST(rp2.pit_number AS TEXT) || '-' ||
        CAST(rp3.pit_number AS TEXT) as pred_combo
    FROM race_predictions rp1
    JOIN race_predictions rp2 ON rp1.race_id = rp2.race_id
        AND rp2.prediction_type = 'before' AND rp2.rank_prediction = 2
    JOIN race_predictions rp3 ON rp1.race_id = rp3.race_id
        AND rp3.prediction_type = 'before' AND rp3.rank_prediction = 3
    WHERE rp1.prediction_type = 'before' AND rp1.rank_prediction = 1
),
confidence_a_races AS (
    SELECT
        r.race_id,
        r.venue_code,
        SUBSTR(r.date, 1, 4) as year,
        pc.pred_combo,
        t.odds,
        -- オッズ帯分類
        CASE
            WHEN t.odds < 10 THEN '<10倍'
            WHEN t.odds < 20 THEN '10-20倍'
            WHEN t.odds < 30 THEN '20-30倍'
            WHEN t.odds < 50 THEN '30-50倍'
            WHEN t.odds < 100 THEN '50-100倍'
            ELSE '100倍+'
        END as odds_band
    FROM races r
    JOIN prediction_combos pc ON r.race_id = pc.race_id
    JOIN trifecta_odds t ON pc.race_id = t.race_id
        AND t.combination = pc.pred_combo
    WHERE pc.confidence_level = 'A'
        AND r.date BETWEEN '2020-01-01' AND '2025-12-31'
)
SELECT
    odds_band,
    COUNT(*) as race_count,
    ROUND(AVG(CASE WHEN pay.payout > 0 THEN 1 ELSE 0 END) * 100, 2) as hit_rate,
    ROUND(AVG(car.odds), 2) as avg_odds,
    ROUND(SUM(COALESCE(pay.payout, 0)) / COUNT(*) * 100, 1) as roi,
    SUM(COALESCE(pay.payout, 0) - 100) as total_profit
FROM confidence_a_races car
LEFT JOIN payouts pay ON car.race_id = pay.race_id
    AND pay.bet_type = 'trifecta'
    AND pay.combination = car.pred_combo
GROUP BY odds_band
ORDER BY
    CASE odds_band
        WHEN '<10倍' THEN 1
        WHEN '10-20倍' THEN 2
        WHEN '20-30倍' THEN 3
        WHEN '30-50倍' THEN 4
        WHEN '50-100倍' THEN 5
        WHEN '100倍+' THEN 6
    END;
```

WITH prediction_combos AS (
    -- (同上)
),
confidence_a_a1_races AS (
    SELECT
        r.race_id,
        pc.pred_combo,
        t.odds,
        e.racer_class
    FROM races r
    JOIN prediction_combos pc ON r.race_id = pc.race_id
    JOIN trifecta_odds t ON pc.race_id = t.race_id
        AND t.combination = pc.pred_combo

        AND e.pit_number = CAST(SUBSTR(pc.pred_combo, 1, 1) AS INTEGER)  -- 1着予測選手

    AND pay.bet_type = 'trifecta'
    AND pay.combination = car.pred_combo;
